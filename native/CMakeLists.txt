cmake_minimum_required(VERSION 3.20)
project(ncnn_agent_bridge LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NCNN_LLM_SOURCE_DIR "" CACHE PATH "Existing, pinned futz12/ncnn_llm source checkout")
set(AGENT_NCNN_SOURCE_DIR "" CACHE PATH "Optional pinned ncnn source; otherwise use find_package")
set(AGENT_JSON_SOURCE_DIR "" CACHE PATH "Optional pinned nlohmann/json source")
if(NOT EXISTS "${NCNN_LLM_SOURCE_DIR}/src/ncnn_llm_gpt.h")
    message(FATAL_ERROR "Set NCNN_LLM_SOURCE_DIR to your existing ncnn_llm checkout. No implicit downloads occur.")
endif()

# Source-build mode is used by CI. The existing installed-SDK mode remains available.
if(AGENT_NCNN_SOURCE_DIR)
    if(NOT EXISTS "${AGENT_NCNN_SOURCE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "AGENT_NCNN_SOURCE_DIR does not contain ncnn sources")
    endif()
    option(NCNN_BUILD_TOOLS "" OFF)
    option(NCNN_BUILD_EXAMPLES "" OFF)
    option(NCNN_BUILD_TESTS "" OFF)
    option(NCNN_BUILD_BENCHMARK "" OFF)
    option(NCNN_INSTALL_SDK "" OFF)
    option(NCNN_VULKAN "" ON)
    option(NCNN_SIMPLEVK "" ON)
    add_subdirectory("${AGENT_NCNN_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/ncnn")
else()
    find_package(ncnn CONFIG REQUIRED)
endif()
if(AGENT_JSON_SOURCE_DIR)
    if(NOT EXISTS "${AGENT_JSON_SOURCE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "AGENT_JSON_SOURCE_DIR does not contain nlohmann/json sources")
    endif()
    set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
    set(JSON_Install OFF CACHE BOOL "" FORCE)
    add_subdirectory("${AGENT_JSON_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/json")
else()
    find_package(nlohmann_json CONFIG QUIET)
endif()
if(TARGET ncnn::ncnn)
    set(NCNN_TARGET ncnn::ncnn)
elseif(TARGET ncnn)
    set(NCNN_TARGET ncnn)
else()
    message(FATAL_ERROR "ncnn package exported no supported target")
endif()
if(TARGET nlohmann_json::nlohmann_json)
    set(JSON_TARGET nlohmann_json::nlohmann_json)
else()
    find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
    add_library(agent_json INTERFACE)
    target_include_directories(agent_json INTERFACE "${NLOHMANN_JSON_INCLUDE_DIR}")
    set(JSON_TARGET agent_json)
endif()
file(GLOB CORE CONFIGURE_DEPENDS "${NCNN_LLM_SOURCE_DIR}/src/*.cpp" "${NCNN_LLM_SOURCE_DIR}/src/utils/*.cpp")
file(GLOB TOKENIZER CONFIGURE_DEPENDS "${NCNN_LLM_SOURCE_DIR}/src/utils/tokenizer/*.cpp")
add_library(agent_ncnn_llm STATIC ${CORE} ${TOKENIZER})
target_include_directories(agent_ncnn_llm PUBLIC "${NCNN_LLM_SOURCE_DIR}/src")
target_link_libraries(agent_ncnn_llm PUBLIC ${NCNN_TARGET} ${JSON_TARGET})
add_executable(ncnn_agent_bridge ncnn_agent_bridge.cpp)
target_link_libraries(ncnn_agent_bridge PRIVATE agent_ncnn_llm)
if(WIN32)
    target_compile_definitions(agent_ncnn_llm PUBLIC NOMINMAX)
    target_link_libraries(agent_ncnn_llm PUBLIC user32 gdi32 shell32)
endif()
if(MSVC)
    target_compile_options(agent_ncnn_llm PUBLIC /utf-8 /EHsc)
    target_compile_options(ncnn_agent_bridge PRIVATE /we4530)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(bridge_cli_tests cli_options_test.cpp)
    add_executable(bridge_exception_tests exception_unwind_test.cpp)
    if(MSVC)
        target_compile_options(bridge_cli_tests PRIVATE /utf-8 /EHsc /we4530)
        target_compile_options(bridge_exception_tests PRIVATE /utf-8 /EHsc /we4530)
    endif()
    add_test(NAME bridge_exception_unwind COMMAND bridge_exception_tests)
    set_tests_properties(bridge_exception_unwind PROPERTIES
        PASS_REGULAR_EXPRESSION "Native unwind cases: 3; failures: 0" TIMEOUT 30)
    add_test(NAME bridge_cli_parse COMMAND bridge_cli_tests)
    set_tests_properties(bridge_cli_parse PROPERTIES TIMEOUT 30)
    add_test(NAME bridge_help COMMAND ncnn_agent_bridge --help)
    set_tests_properties(bridge_help PROPERTIES PASS_REGULAR_EXPRESSION "Usage: ncnn_agent_bridge" TIMEOUT 30)
    add_test(NAME bridge_version COMMAND ncnn_agent_bridge --version)
    set_tests_properties(bridge_version PROPERTIES PASS_REGULAR_EXPRESSION "ncnn_agent_bridge 0.1.0" TIMEOUT 30)
endif()

# Version-matched hardware probe distributed beside the text engine.
add_executable(ncnn_device_probe device_probe.cpp)
target_link_libraries(ncnn_device_probe PRIVATE ${NCNN_TARGET})
if(MSVC)
    target_compile_options(ncnn_device_probe PRIVATE /utf-8 /EHsc)
endif()
add_dependencies(ncnn_agent_bridge ncnn_device_probe)
