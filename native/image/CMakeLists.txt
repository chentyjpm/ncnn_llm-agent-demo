cmake_minimum_required(VERSION 3.20)
project(local_agent_image LANGUAGES C CXX)
set(QWENIMAGE_SOURCE_DIR "" CACHE PATH "Pinned Qwen Image source checkout")
if(NOT EXISTS "${QWENIMAGE_SOURCE_DIR}/src/CMakeLists.txt")
    message(FATAL_ERROR "Set QWENIMAGE_SOURCE_DIR to the pinned Qwen Image source")
endif()
add_subdirectory("${QWENIMAGE_SOURCE_DIR}/src" upstream)
add_executable(ncnn_device_probe ../device_probe.cpp)
target_compile_features(ncnn_device_probe PRIVATE cxx_std_11)
target_link_libraries(ncnn_device_probe PRIVATE ncnn)
if(MSVC)
    target_compile_options(ncnn_device_probe PRIVATE /utf-8 /EHsc)
endif()
set_target_properties(qwenimage-ncnn-vulkan PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_dependencies(qwenimage-ncnn-vulkan ncnn_device_probe)
